const builders = [
    {
        id: 1,
        developer: "Fairgate Homes",
        name: "Upper Stouffville",
        location: "Whitchurch-Stouffville, Ontario",
        address: "Baker Hill Boulevard & West Lawn Crescent, L4A 7X3",
        type: "Single Family Home",
        conStatus: "Construction",
        ceilings: "8 feet",

        features: [
            "Double car garage",
            "Open concept kitchen",
            "Large primary bedroom",
            "Modern exterior",
            "Family-friendly community"
        ]
    }
]

export default function BuilderPage({ params }: { params: { id: string} }) {
    //find the builder whose ID matches the ID in the URL
    //params.id pulls the ID from the url, Numbers(params.id) converts it to an int since the data uses ints
    //builders.find finds the first item that matches the condition, the condition in this case being that b.id === 2
    const builder = builders.find((b) => b.id === Number(params.id))

    if (!builder) return notFound();

    return (
        <main className="p-10">
            <img
                src={builder.img}
                className="w-full max-h-[400px] object-cover rounded-lg mb-6"
            />

            <h1 className="text-3x1 font-bold mb-2">
                {builder.name}
            </h1>

            <p className="text-x1 mb-4">
                {/* toLocaleString formats things in a human friendly way, i.e: 2000000 -> 2,000,000 */}
                {builder.price.toLocaleString()}
            </p>

            <p className="mb-4">
                {builder.desc}
            </p>

            <p className="mb-4">
                {builder.beds} Bed • {builder.baths} Bath • {builder.sqft} sqft
            </p>

            <h1 className="font-semibold mb-2">
                Features
            </h1>

            <ul className="list-disc pl-5">
                {builder.features.map((f, i) => (
                    <li key={i}>{f}</li>
                ))}
            </ul>
        </main>
    )
}
